# cleaning
cleaned_salaries = salaries %>% 
  pivot_longer(cols = c('AvgFullProfSalary', 'AvgAssocProfSalary', 'AvgAssistProfSalary'),
               names_to = 'Rank',
               values_to = 'Salary') %>% 
  mutate(Rank = recode(Rank,
                       'AvgFullProfSalary' = 'Full',
                       'AvgAssocProfSalary' = 'Assoc',
                       'AvgAssistProfSalary' = 'Assist'))
